## Usage Examples
The CUDA example (`examples/quickstart/load/load_cuda.cc`) demonstrates the end-to-end pattern for C++ CUDA kernel loading:

```cpp
// Custom CUDA allocator satisfying the NDAlloc concept
struct CUDANDAlloc {
    void AllocData(DLTensor* tensor) { cudaMalloc(&tensor->data, GetDataSize(*tensor)); }
    void FreeData(DLTensor* tensor)  { cudaFree(tensor->data); tensor->data = nullptr; }
};

// Allocate CUDA tensors
ffi::Tensor x = ffi::Tensor::FromNDAlloc(CUDANDAlloc(), {5}, f32_dtype, cuda_device);
ffi::Tensor y = ffi::Tensor::FromNDAlloc(CUDANDAlloc(), {5}, f32_dtype, cuda_device);

// Load and invoke a CUDA kernel from a shared library
ffi::Module mod = ffi::Module::LoadFromFile("build/add_one_cuda.so");
ffi::Function add_one_cuda = mod->GetFunction("add_one_cuda").value();
add_one_cuda(x, y);
```
